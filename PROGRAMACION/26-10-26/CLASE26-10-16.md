			A > B MAYOR QUE

			A >= B MAYOR O IGUAL
			A == B IGUAL
			A != B DISTINTO
			(A Y B PUEDEN SER CUALQUIER EXPRESION)

			int a = 3;

			System.out.print("Es impar");

Ejercicio: UN NUMERO ES DIVISIBLE ENTRE OTRO CUANDO EL RESTO DE LA DIVISION ES CERO.
POR EJEMPLO, 15 ES DIVISIBLE ENTRE 3 PORQUE 15 % 3 == 0.
HACER UN PROGRAMA QUE SOLICITE UN NUMERO AL USUARIO. SI EL NUMERO ES DIVISIBLE ENTRE 3, SE DEBE ESCRIBIR POR PANTALLA "FIZZ". SI EL NUMERO ES DIVISIBLE ENTRE 5, DEBE ESCRIBIR "BUZZ". SI EL NUMERO ES DIVISIBLE ENTRE 15 DEBE ESCRIBIR "FIZZBUZZ". SI NO DEBE ESCRIBIR EL NUMERO.

SOLUCION
import java.util.Scanner;

class Ejercicio1 {
	public static void main(String arg[]) {

		Scanner inputTeclado = new Scanner(System.in);

		System.out.print("Introduce un numero y te dire si es divisible entre 2 5 o 15 con fizz y buzz> ");
		
		int n1 = Integer.parseInt(inputTeclado.nextLine().trim());

		if(n1 % 15 == 0) {
			System.out.print("fizzbuzz");
		} else if(n1 % 5 == 0) {
			System.out.print("buzz");
		} else if(n1 % 3 == 0) {
			System.out.print("fizz");
		} else {
			System.out.print(n1);
		}
	}
}

4) OPERADORES LOGICOS

SE USAN EN SITUACIONES EN LAS QUE SE NECESITAN QUE SE CUMPLAN VARIAS CONDICIONES A LA VEZ O SOLO UNA  DE VARIAS.


OPERADOR && (y lógico [AND]): SE USAN EN SITUACIONES EN LAS QUE SE NECESITAN QUE SE CUMPLAN VARIAS CONDICIONES A LA VEZ

EJEMPLO:

int a = 3;
int b = 4;
if (a < 7 && b > 2) {
	System.out.println("Se cumplen las condiciones");
}
if (a < 7 && b > 2 && (a+b) == 7 && (a-b) == -1 && b == 4) {
	System.out.println("Se cumplen las condiciones");
}

OPERADOR || (o lógico [OR]): SE USAN EN SITUACIONES EN LAS QUE SE NECESITAN QUE SE CUMPLA AL MENOS UNA CONDICIÓN

EJEMPLO:

int a = 3;
int b = 4;
if (a < 7 || b > 2) {
	System.out.println("Se cumplen alguna condicion");
}
if (a < 7 || b > 2 || (a+b) == 7 || (a-b) == -1 || b == 4) {
	System.out.println("Se cumplen alguna condicion");
}

SE PUEDEN MEZCLAR

EJEMPLO:

int a = 3;
int b = 4;
if (a < 7 || b > 2 && (a+b) == 7 || (a-b) == -1 && b == 4) {
	System.out.println("Se cumplen alguna condicion");
}

IMPORTANTE: && TIENE PRIORIDAD SOBRE ||.
